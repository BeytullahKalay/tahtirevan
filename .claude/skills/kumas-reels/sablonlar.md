## 6 · Montaj betiği (`6-montaj/montaj.sh`, reels-N klasöründen `bash 6-montaj/montaj.sh`)
```bash
#!/usr/bin/env bash
set -e
export PATH="/c/Users/Beytullah KALAY/AppData/Local/Microsoft/WinGet/Packages/Gyan.FFmpeg_Microsoft.Winget.Source_8wekyb3d8bbwe/ffmpeg-9.0.2-full_build/bin:$PATH"
K=5-klipler; A=6-montaj/ara; V=v1
mkdir -p $A 6-montaj/cikti
ENC="-an -r 24 -c:v libx264 -crf 10 -preset medium -pix_fmt yuv420p -color_range tv -colorspace bt709 -color_primaries bt709 -color_trc bt709"
GEOM="scale=1080:1920:flags=lanczos,setsar=1"
# parca <çıkış> <klip> <ilk kare> <kare sayısı> [ek süzgeç]
parca() { local extra=""; [ -n "$5" ] && extra=",$5"
  ffmpeg -v error -y -i "$K/$2" -vf "trim=start_frame=$3,setpts=PTS-STARTPTS,$GEOM$extra" -frames:v "$4" $ENC "$A/$1.mp4"; }
# yaklas <giriş> <çıkış> <hedef x> <hedef y> <yakınlaşma 0.08> <süre sn>
yaklas() { local Z="(1+$5*(in/24)/$6)"
  ffmpeg -v error -y -i "$1" -vf "perspective=x0='$3-$3/$Z':y0='$4-$4/$Z':x1='$3+(W-$3)/$Z':y1='$4-$4/$Z':x2='$3-$3/$Z':y2='$4+(H-$4)/$Z':x3='$3+(W-$3)/$Z':y3='$4+(H-$4)/$Z':interpolation=cubic:eval=frame" $ENC "$2"; }

parca 01 c1.mp4 24 60
parca 02 c2.mp4 24 60
# ... her çekim; toplam gövde kare sayısını hesapla (B)
# Kart: 6-montaj/kart/kart.mp4 (2 sn, 48 kare). xfade offset = B/24 - 0.5. Toplam = B + 48 - 12.
B=360; T=$((B+36)); OFF=$(echo "$B/24-0.5" | bc -l)
ffmpeg -v error -y -i $A/01.mp4 -i $A/02.mp4 [...] -i 6-montaj/kart/kart.mp4 -ss [MÜZİK BAŞI] -i 6-montaj/muzik.mp3 \
  -filter_complex "[0][1][...]concat=n=[N]:v=1:a=0,fps=24,settb=AVTB,format=yuv420p[body];[N]fps=24,settb=AVTB,format=yuv420p[card];[body][card]xfade=transition=fade:duration=0.5:offset=$OFF,noise=c0s=5:c0f=t+u,format=yuv420p[v];[N+1]atrim=0:$(echo "$T/24"|bc -l),asetpts=PTS-STARTPTS,afade=t=in:d=0.3,afade=t=out:st=[SON-1.2]:d=1.2[a]" \
  -map "[v]" -map "[a]" -frames:v $T -r 24 -c:v libx264 -profile:v high -crf 16 -preset slow -pix_fmt yuv420p \
  -color_range tv -colorspace bt709 -color_primaries bt709 -color_trc bt709 -c:a aac -b:a 192k -movflags +faststart \
  6-montaj/cikti/reels-N-$V.mp4
```
Her yeni sürüm yeni dosya (`montaj-v2.sh` → `reels-N-v2.mp4`); eskiler silinmez.
